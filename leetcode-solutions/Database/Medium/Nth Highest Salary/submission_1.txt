# Problem: Nth Highest Salary
# Difficulty: Medium
# Topic: Database
# Language: mysql
# Submission: 1

CREATE FUNCTION getNthHighestSalary(N INT) RETURNS INT
BEGIN
  DECLARE toprows INT;
  SET toprows = N - 1;

  RETURN (
      SELECT DISTINCT salary
      FROM Employee
      ORDER BY salary DESC
      LIMIT toprows, 1
  );
END;
